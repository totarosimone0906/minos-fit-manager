        />

          testo="Chat"

        <BottomNavButton
          testo="Notifiche"
          icona={<BellIcon />}

function SchedaSecondaria({
  sezione,
  cliente,
}: {
  sezione: Sezione
  cliente: {
    scadenzaAbbonamento: string
    scadenzaQuota: string
    scadenzaCertificato: string
  }
}) {
  let descrizione = ""
  let pulsante = ""

  if (sezione.id === "abbonamento") {
    descrizione = `Scadenza abbonamento: ${cliente.scadenzaAbbonamento}`
    pulsante = "Rinnova abbonamento"
  }

  if (sezione.id === "quota") {
    descrizione = `Scadenza quota associativa: ${cliente.scadenzaQuota}`
    pulsante = "Rinnova quota associativa"
  }

  if (sezione.id === "certificato") {
    descrizione = `Scadenza certificato: ${cliente.scadenzaCertificato}`
    pulsante = "Carica certificato"
  }

  if (sezione.id === "prenotazioni") {
    descrizione = "Visualizza disponibilità e prenota il tuo personal trainer."
    pulsante = "Apri calendario PT"
  }

  if (sezione.id === "schede") {
    descrizione = "Consulta la tua scheda di allenamento."
    pulsante = "Apri la mia scheda"
  }

  if (sezione.id === "ia") {
    descrizione = "Parla con l’assistente virtuale MinosFit."
    pulsante = "Apri assistente IA"
  }

  if (sezione.id === "contatti") {
    descrizione = "Contatta la palestra o il tuo personal trainer."
    pulsante = "Apri contatti"
  }

  return (
    <article className="flex min-h-[390px] w-full flex-col rounded-[32px] border border-zinc-700 bg-gradient-to-br from-[#151617] to-[#090a0b] p-8 shadow-[0_35px_80px_rgba(0,0,0,0.75)]">
      <div className="flex items-center gap-4 text-red-500">
        <span className="flex h-14 w-14 items-center justify-center rounded-full border border-red-600/50 bg-red-950/20">
          {sezione.icona}
        </span>

        <div>
          <p className="text-xs font-black uppercase tracking-[0.2em] text-zinc-500">
            Area cliente
          </p>

          <h2 className="mt-1 text-3xl font-black text-white">
            {sezione.nome}
          </h2>
        </div>
      </div>

      <div className="mt-9 rounded-2xl border border-zinc-800 bg-black/40 p-6">
        <p className="text-sm leading-7 text-zinc-300">{descrizione}</p>
      </div>

      <button
        type="button"
        className="mt-auto rounded-2xl bg-red-600 px-6 py-4 text-sm font-black uppercase tracking-wide transition hover:bg-red-500"
      >
        {pulsante}
      </button>
    </article>
  )
}

function RigaStato({
  icona,
  testo,
  stato,
}: {
  icona: ReactNode
  testo: string
  stato: Stato
}) {
  return (
    <div className="flex items-center gap-3">
      <span className="flex h-10 w-10 shrink-0 items-center justify-center rounded-full border border-zinc-700 bg-black/35 text-red-500">
        {icona}
      </span>

      <span className="min-w-0 flex-1 text-sm font-bold text-zinc-100 sm:text-base">
        {testo}
      </span>

      <BadgeStato stato={stato} />
    </div>
  )
}

function BadgeStato({ stato }: { stato: Stato }) {
  let stile = "border-zinc-600 text-zinc-400"

  if (stato === "ATTIVO") {
    stile =
      "border-emerald-500 text-emerald-400 shadow-[0_0_16px_rgba(16,185,129,0.13)]"
  }

  if (stato === "IN SCADENZA") {
    stile =
      "border-orange-500 text-orange-400 shadow-[0_0_16px_rgba(249,115,22,0.13)]"
  }

  if (stato === "SCADUTO") {
    stile =
      "border-red-600 text-red-500 shadow-[0_0_16px_rgba(220,38,38,0.13)]"
  }

  return (
    <span
      className={`inline-flex min-w-[92px] items-center justify-center rounded-full border px-3 py-2 text-center text-[10px] font-black uppercase tracking-wide sm:min-w-[112px] sm:text-xs ${stile}`}
    >
      {stato}
    </span>
  )
}
